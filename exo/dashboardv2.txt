🟦 1) Permission API (.NET 8) – pełny projekt

Program.cs

using System.DirectoryServices.AccountManagement;
using Microsoft.AspNetCore.Server.IISIntegration;

var builder = WebApplication.CreateBuilder(args);

// Windows Authentication
builder.Services.AddAuthentication(IISDefaults.AuthenticationScheme);
builder.Services.AddAuthorization();

var app = builder.Build();

app.UseAuthentication();
app.UseAuthorization();

app.MapGet("/api/permissions", (HttpContext ctx) =>
{
    var user = ctx.User.Identity?.Name;
    if (string.IsNullOrEmpty(user))
        return Results.Unauthorized();

    var domain = user.Split('\\')[0];
    var username = user.Split('\\')[1];

    var permissions = PermissionEvaluator.GetPermissions(domain, username);

    return Results.Json(permissions);
})
.RequireAuthorization();

app.Run();

PermissionEvaluator.cs

using System.DirectoryServices.AccountManagement;

public static class PermissionEvaluator
{
    public static Dictionary<string, bool> GetPermissions(string domain, string username)
    {
        var result = new Dictionary<string, bool>
        {
            ["usage"] = false,
            ["usage_summary"] = false,
            ["usage_runjobs"] = false,
            ["usage_logs"] = false,

            ["exo_tools"] = false,
            ["exo_tool1"] = false,
            ["exo_tool2"] = false,
            ["exo_tool3"] = false,

            ["teams_tools"] = false
        };

        using var ctx = new PrincipalContext(ContextType.Domain, domain);
        using var user = UserPrincipal.FindByIdentity(ctx, username);

        if (user == null)
            return result;

        bool InGroup(string groupName)
        {
            return user.GetAuthorizationGroups()
                       .Any(g => g.SamAccountName.Equals(groupName, StringComparison.OrdinalIgnoreCase));
        }

        // MAPOWANIE GRUP AD → UPRAWNIENIA
        if (InGroup("APP-Usage-Reader"))
        {
            result["usage"] = true;
            result["usage_summary"] = true;
            result["usage_logs"] = true;
        }

        if (InGroup("APP-Usage-Admin"))
        {
            result["usage"] = true;
            result["usage_summary"] = true;
            result["usage_logs"] = true;
            result["usage_runjobs"] = true;
        }

        if (InGroup("APP-EXO-Tools"))
        {
            result["exo_tools"] = true;
            result["exo_tool1"] = true;
            result["exo_tool2"] = true;
            result["exo_tool3"] = true;
        }

        if (InGroup("APP-Teams-Tools"))
        {
            result["teams_tools"] = true;
        }

        return result;
    }
}

js/permissions.js

async function loadPermissions() {
    const response = await fetch("/api/permissions");
    return await response.json();
}

async function buildMenu() {
    const permissions = await loadPermissions();
    const root = document.getElementById("menu-root");

    MenuConfig.forEach(section => {
        if (!permissions[section.permission]) return;

        const li = document.createElement("li");
        li.classList.add("menu-section");

        const title = document.createElement("div");
        title.classList.add("menu-title");
        title.textContent = section.title;

        const submenu = document.createElement("ul");
        submenu.classList.add("submenu");

        section.children.forEach(child => {
            if (!permissions[child.permission]) return;

            const item = document.createElement("li");
            item.textContent = child.title;
            item.dataset.view = child.id;
            submenu.appendChild(item);
        });

        li.appendChild(title);
        li.appendChild(submenu);
        root.appendChild(li);
    });
}

buildMenu();

js/menu.js

document.addEventListener("click", e => {
    if (!e.target.classList.contains("menu-title")) return;

    const section = e.target.parentElement;

    document.querySelectorAll(".menu-section").forEach(s => {
        if (s !== section) s.classList.remove("open");
    });

    section.classList.toggle("open");
});


---

js/router.js

document.addEventListener("click", async e => {
    if (!e.target.dataset.view) return;

    const view = e.target.dataset.view;

    const response = await fetch(`/views/${view}.html`);
    const html = await response.text();
    document.getElementById("content").innerHTML = html;

    document.querySelectorAll(".submenu li").forEach(li => li.classList.remove("active-item"));
    document.querySelectorAll(".menu-section").forEach(sec => sec.classList.remove("active-section"));

    e.target.classList.add("active-item");

    const section = e.target.closest(".menu-section");
    section.classList.add("active-section");
    section.classList.add("open");
});
