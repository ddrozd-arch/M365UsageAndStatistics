⚙️ js/menu.js

(Obsługa rozwijania sekcji)

document.querySelectorAll(".menu-title").forEach(title => {
    title.addEventListener("click", () => {
        const section = title.parentElement;

        // Akordeon – zamyka inne sekcje
        document.querySelectorAll(".menu-section").forEach(s => {
            if (s !== section) s.classList.remove("open");
        });

        section.classList.toggle("open");
    });
});


---

🔄 js/router.js

(Ładowanie widoków + podświetlanie aktywnej sekcji i elementu)

document.querySelectorAll("[data-view]").forEach(item => {
    item.addEventListener("click", async () => {
        const view = item.getAttribute("data-view");

        // Ładowanie widoku
        const response = await fetch(`/views/${view}.html`);
        const html = await response.text();
        document.getElementById("content").innerHTML = html;

        // Reset podświetleń
        document.querySelectorAll(".submenu li").forEach(li => li.classList.remove("active-item"));
        document.querySelectorAll(".menu-section").forEach(sec => sec.classList.remove("active-section"));

        // Ustaw aktywny element
        item.classList.add("active-item");

        // Ustaw aktywną sekcję
        const section = item.closest(".menu-section");
        section.classList.add("active-section");
        section.classList.add("open");
    });
});
