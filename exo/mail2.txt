Poniżej masz rozszerzoną wersję klasy, która obsługuje teraz:

• CC
• BCC
• alias nadawcy
• HTML body
• trzy tryby wysyłki (anonimowy, default credentials, authenticated)


Dodałem parametry:

• IEnumerable<string> cc = null
• IEnumerable<string> bcc = null


Wszystkie metody je przyjmują i przekazują do BuildMessage.

---

✨ Klasa z obsługą CC / BCC

using System;
using System.Collections.Generic;
using System.IO;
using System.Net;
using System.Net.Mail;

public class SmtpMailSender
{
    private readonly string _smtpHost;
    private readonly int _smtpPort;
    private readonly bool _enableSsl;

    public SmtpMailSender(string smtpHost, int smtpPort = 587, bool enableSsl = true)
    {
        _smtpHost = smtpHost;
        _smtpPort = smtpPort;
        _enableSsl = enableSsl;
    }

    // ============================================================
    // 1) Wysyłanie anonimowe
    // ============================================================
    public void SendAnonymous(
        IEnumerable<string> recipients,
        string subject,
        string body = "",
        string attachmentPath = null,
        string fromAddress = null,
        string fromAlias = null,
        bool isHtml = false,
        IEnumerable<string> cc = null,
        IEnumerable<string> bcc = null)
    {
        var client = BuildClientAnonymous();
        var msg = BuildMessage(fromAddress, fromAlias, recipients, subject, body, attachmentPath, isHtml, cc, bcc);
        client.Send(msg);
    }

    // ============================================================
    // 2) Wysyłanie jako Default Credentials
    // ============================================================
    public void SendAsDefaultCredentials(
        string fromAddress,
        IEnumerable<string> recipients,
        string subject,
        string body = "",
        string attachmentPath = null,
        string fromAlias = null,
        bool isHtml = false,
        IEnumerable<string> cc = null,
        IEnumerable<string> bcc = null)
    {
        var client = BuildClientDefaultCredentials();
        var msg = BuildMessage(fromAddress, fromAlias, recipients, subject, body, attachmentPath, isHtml, cc, bcc);
        client.Send(msg);
    }

    // ============================================================
    // 3) Wysyłanie z loginem/hasłem
    // ============================================================
    public void SendAuthenticated(
        string fromAddress,
        NetworkCredential credentials,
        IEnumerable<string> recipients,
        string subject,
        string body = "",
        string attachmentPath = null,
        string fromAlias = null,
        bool isHtml = false,
        IEnumerable<string> cc = null,
        IEnumerable<string> bcc = null)
    {
        var client = BuildClientAuthenticated(credentials);
        var msg = BuildMessage(fromAddress, fromAlias, recipients, subject, body, attachmentPath, isHtml, cc, bcc);
        client.Send(msg);
    }

    // ============================================================
    // PRIVATE HELPERS
    // ============================================================

    private SmtpClient BuildClientAnonymous()
    {
        return new SmtpClient(_smtpHost)
        {
            Port = _smtpPort,
            EnableSsl = _enableSsl,
            UseDefaultCredentials = false,
            Credentials = null
        };
    }

    private SmtpClient BuildClientDefaultCredentials()
    {
        return new SmtpClient(_smtpHost)
        {
            Port = _smtpPort,
            EnableSsl = _enableSsl,
            UseDefaultCredentials = true,
            Credentials = CredentialCache.DefaultNetworkCredentials
        };
    }

    private SmtpClient BuildClientAuthenticated(NetworkCredential credentials)
    {
        return new SmtpClient(_smtpHost)
        {
            Port = _smtpPort,
            EnableSsl = _enableSsl,
            UseDefaultCredentials = false,
            Credentials = credentials
        };
    }

    private MailMessage BuildMessage(
        string fromAddress,
        string fromAlias,
        IEnumerable<string> recipients,
        string subject,
        string body,
        string attachmentPath,
        bool isHtml,
        IEnumerable<string> cc,
        IEnumerable<string> bcc)
    {
        var msg = new MailMessage
        {
            Subject = subject,
            Body = body ?? "",
            IsBodyHtml = isHtml
        };

        // FROM z aliasem
        if (!string.IsNullOrWhiteSpace(fromAddress))
        {
            msg.From = string.IsNullOrWhiteSpace(fromAlias)
                ? new MailAddress(fromAddress)
                : new MailAddress(fromAddress, fromAlias);
        }

        // TO
        foreach (var r in recipients)
            msg.To.Add(r);

        // CC
        if (cc != null)
            foreach (var c in cc)
                msg.CC.Add(c);

        // BCC
        if (bcc != null)
            foreach (var b in bcc)
                msg.Bcc.Add(b);

        // Załącznik
        if (!string.IsNullOrWhiteSpace(attachmentPath) && File.Exists(attachmentPath))
            msg.Attachments.Add(new Attachment(attachmentPath));

        return msg;
    }
}

🔧 Przykład użycia CC / BCC

✔ Default Credentials + CC + BCC + HTML

mailer.SendAsDefaultCredentials(
    fromAddress: "service@domain.com",
    fromAlias: "System Powiadomień",
    recipients: new[] { "user@domain.com" },
    cc: new[] { "cc@domain.com" },
    bcc: new[] { "audyt@domain.com" },
    subject: "Raport",
    body: "<h2>Raport dzienny</h2><p>Wszystko OK</p>",
    isHtml: true
);
