Poniżej masz uniwersalną, gotową do użycia klasę SmtpMailSender, która obsługuje:

• wysyłanie anonimowe (bez logowania, np. relay w sieci LAN)
• wysyłanie z DefaultNetworkCredentials (np. Windows Service działający na koncie domenowym)
• wysyłanie z dowolnymi poświadczeniami (opcjonalnie)


Klasa przyjmuje parametry:

• temat
• body (opcjonalnie)
• listę adresatów
• załącznik (opcjonalnie, jako string path lub Stream)


---

✨ Uniwersalna klasa SMTP

using System;
using System.Collections.Generic;
using System.IO;
using System.Net;
using System.Net.Mail;

public class SmtpMailSender
{
    private readonly string _smtpHost;
    private readonly int _smtpPort;
    private readonly bool _enableSsl;

    public SmtpMailSender(string smtpHost, int smtpPort = 587, bool enableSsl = true)
    {
        _smtpHost = smtpHost;
        _smtpPort = smtpPort;
        _enableSsl = enableSsl;
    }

    // -----------------------------
    // 1) Wysyłanie anonimowe
    // -----------------------------
    public void SendAnonymous(
        IEnumerable<string> recipients,
        string subject,
        string body = "",
        string attachmentPath = null)
    {
        var client = BuildClientAnonymous();
        var msg = BuildMessage(null, recipients, subject, body, attachmentPath);
        client.Send(msg);
    }

    // -----------------------------
    // 2) Wysyłanie jako Default Credentials
    // -----------------------------
    public void SendAsDefaultCredentials(
        string from,
        IEnumerable<string> recipients,
        string subject,
        string body = "",
        string attachmentPath = null)
    {
        var client = BuildClientDefaultCredentials();
        var msg = BuildMessage(from, recipients, subject, body, attachmentPath);
        client.Send(msg);
    }

    // -----------------------------
    // 3) Wysyłanie z loginem/hasłem
    // -----------------------------
    public void SendAuthenticated(
        string from,
        NetworkCredential credentials,
        IEnumerable<string> recipients,
        string subject,
        string body = "",
        string attachmentPath = null)
    {
        var client = BuildClientAuthenticated(credentials);
        var msg = BuildMessage(from, recipients, subject, body, attachmentPath);
        client.Send(msg);
    }

    // ============================================================
    // PRIVATE HELPERS
    // ============================================================

    private SmtpClient BuildClientAnonymous()
    {
        return new SmtpClient(_smtpHost)
        {
            Port = _smtpPort,
            EnableSsl = _enableSsl,
            UseDefaultCredentials = false,
            Credentials = null
        };
    }

    private SmtpClient BuildClientDefaultCredentials()
    {
        return new SmtpClient(_smtpHost)
        {
            Port = _smtpPort,
            EnableSsl = _enableSsl,
            UseDefaultCredentials = true,
            Credentials = CredentialCache.DefaultNetworkCredentials
        };
    }

    private SmtpClient BuildClientAuthenticated(NetworkCredential credentials)
    {
        return new SmtpClient(_smtpHost)
        {
            Port = _smtpPort,
            EnableSsl = _enableSsl,
            UseDefaultCredentials = false,
            Credentials = credentials
        };
    }

    private MailMessage BuildMessage(
        string from,
        IEnumerable<string> recipients,
        string subject,
        string body,
        string attachmentPath)
    {
        var msg = new MailMessage
        {
            Subject = subject,
            Body = body ?? "",
            From = from != null ? new MailAddress(from) : null
        };

        foreach (var r in recipients)
            msg.To.Add(r);

        if (!string.IsNullOrWhiteSpace(attachmentPath) && File.Exists(attachmentPath))
            msg.Attachments.Add(new Attachment(attachmentPath));

        return msg;
    }
}


---

🔧 Jak używać

✔ Wysyłanie anonimowe (relay LAN)

var mailer = new SmtpMailSender("smtp.local");
mailer.SendAnonymous(
    new[] { "user@domain.com" },
    "Test anonimowy",
    "Treść wiadomości"
);


✔ Wysyłanie jako użytkownik domenowy (Windows Service)

var mailer = new SmtpMailSender("smtp.domain.com");
mailer.SendAsDefaultCredentials(
    "serviceuser@domain.com",
    new[] { "odbiorca@domain.com" },
    "Test",
    "Wiadomość z Windows Service"
);


✔ Wysyłanie z loginem/hasłem

var mailer = new SmtpMailSender("smtp.domain.com");

mailer.SendAuthenticated(
    "noreply@domain.com",
    new NetworkCredential("noreply@domain.com", "P@ssw0rd"),
    new[] { "odbiorca@domain.com" },
    "Test",
    "Treść"
);
