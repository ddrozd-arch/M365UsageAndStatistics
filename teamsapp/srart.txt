🟦 TEAMSAPP — Program.cs (z Windows Auth + CorrelationId)

using Microsoft.AspNetCore.Authentication.Negotiate;
using Microsoft.Extensions.Options;
using System.Text.Json;

var builder = WebApplication.CreateBuilder(args);

// Windows Authentication
builder.Services.AddAuthentication(NegotiateDefaults.AuthenticationScheme)
    .AddNegotiate();

builder.Services.AddAuthorization();

// Load config with reloadOnChange
builder.Configuration.AddJsonFile("appsettings.json", optional: false, reloadOnChange: true);

// Bind config
builder.Services.Configure<AppConfig>(builder.Configuration.GetSection("AppConfig"));

// HttpClient factory
builder.Services.AddHttpClient();

// PermissionService client
builder.Services.AddHttpClient("PermissionService", client =>
{
    var url = builder.Configuration["PermissionService:Url"];
    client.BaseAddress = new Uri(url);
});

// Graph placeholder
builder.Services.AddSingleton<GraphHttpClient>();

var app = builder.Build();

// 🔵 CorrelationId middleware
app.Use(async (context, next) =>
{
    const string headerName = "X-Correlation-Id";

    if (!context.Request.Headers.TryGetValue(headerName, out var cid) || string.IsNullOrWhiteSpace(cid))
    {
        cid = Guid.NewGuid().ToString();
        context.Request.Headers[headerName] = cid;
    }

    context.Response.Headers[headerName] = cid;
    context.Items["CorrelationId"] = cid.ToString();

    await next();
});

app.UseAuthentication();
app.UseAuthorization();

// Map APIs
ConfigApi.Map(app);
RoleApi.Map(app);

app.Run();

🟦 TEAMSAPP — ConfigApi.cs

using System.Text.Json;

public static class ConfigApi
{
    public static void Map(WebApplication app)
    {
        // GET /config
        app.MapGet("/config", (IOptionsMonitor<AppConfig> cfg) =>
        {
            return Results.Ok(cfg.CurrentValue);
        });

        // POST /config/save
        app.MapPost("/config/save", async (AppConfig newCfg) =>
        {
            var json = JsonSerializer.Serialize(
                new { AppConfig = newCfg },
                new JsonSerializerOptions { WriteIndented = true }
            );

            await File.WriteAllTextAsync("appsettings.json", json);
            return Results.Ok(new { status = "saved" });
        });

        // POST /config/validate
        app.MapPost("/config/validate", (AppConfig cfg) =>
        {
            if (string.IsNullOrWhiteSpace(cfg.AppId)) return Results.BadRequest("AppId missing");
            if (string.IsNullOrWhiteSpace(cfg.Secret)) return Results.BadRequest("Secret missing");
            if (string.IsNullOrWhiteSpace(cfg.TenantId)) return Results.BadRequest("TenantId missing");
            if (string.IsNullOrWhiteSpace(cfg.Thumbprint)) return Results.BadRequest("Thumbprint missing");

            return Results.Ok(new { valid = true });
        });
    }
}

🟦 TEAMSAPP — RoleApi.cs (z przekazywaniem CorrelationId)

public static class RoleApi
{
    public static void Map(WebApplication app)
    {
        app.MapGet("/roles", async (HttpContext ctx, IHttpClientFactory factory) =>
        {
            var client = factory.CreateClient("PermissionService");

            const string headerName = "X-Correlation-Id";
            var cid = ctx.Request.Headers[headerName].FirstOrDefault();

            if (!string.IsNullOrWhiteSpace(cid))
                client.DefaultRequestHeaders.Add(headerName, cid);

            var response = await client.GetAsync("");
            if (!response.IsSuccessStatusCode)
                return Results.StatusCode((int)response.StatusCode);

            var roles = await response.Content.ReadFromJsonAsync<RoleResponse>();
            return Results.Ok(roles);
        });
    }
}

public class RoleResponse
{
    public string CorrelationId { get; set; }
    public string User { get; set; }
    public List<string> Roles { get; set; }
    public List<string> Groups { get; set; }
    public int Ttl { get; set; }
}

🟩 PERMISSIONSERVICE — Program.cs (Windows Auth + CorrelationId)

using Microsoft.AspNetCore.Authentication.Negotiate;

var builder = WebApplication.CreateBuilder(args);

builder.Services.AddAuthentication(NegotiateDefaults.AuthenticationScheme)
    .AddNegotiate();

builder.Services.AddAuthorization();

builder.Services.AddSingleton<IAdGroupResolver, AdGroupResolver>();

var app = builder.Build();

app.UseAuthentication();
app.UseAuthorization();

app.MapGet("/api/roles", async (HttpContext ctx, IAdGroupResolver ad, ILogger<Program> logger) =>
{
    const string headerName = "X-Correlation-Id";
    var cid = ctx.Request.Headers[headerName].FirstOrDefault() ?? Guid.NewGuid().ToString();

    var user = ctx.User.Identity?.Name ?? "unknown";

    logger.LogInformation("CorrelationId={CorrelationId} User={User} - fetching roles", cid, user);

    var groups = await ad.GetGroupsAsync(user);

    var roles = groups.Contains("CONTOSO\\App_Admins")
        ? new List<string> { "Admin" }
        : new List<string> { "User" };

    return Results.Ok(new {
        correlationId = cid,
        user,
        roles,
        groups,
        ttl = 300
    });
});

app.Run();
